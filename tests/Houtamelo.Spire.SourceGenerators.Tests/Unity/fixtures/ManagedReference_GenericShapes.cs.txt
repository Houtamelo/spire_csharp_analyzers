#nullable enable
using System;
using System.Collections.Generic;
using Houtamelo.Spire;
using UnityEngine;

[Serializable]
public abstract class ManagedReferenceGenericBase
{
    public int baseValue;
}

public interface ManagedReferenceGenericInterface
{
}

[Serializable]
public sealed class ManagedReferenceGenericLeaf : ManagedReferenceGenericBase, ManagedReferenceGenericInterface
{
    public string label = string.Empty;
}

[Serializable]
[DiscriminatedUnion(Layout.Additive)]
public partial struct ManagedReferenceGenericUnion<T>
{
    [Variant]
    public static partial ManagedReferenceGenericUnion<T> Generic(
        [UnitySerializeReference] T genericValue);

    [Variant]
    public static partial ManagedReferenceGenericUnion<T> Base(
        [UnitySerializeReference] ManagedReferenceGenericBase baseValue);

    [Variant]
    public static partial ManagedReferenceGenericUnion<T> Interface(
        [UnitySerializeReference] ManagedReferenceGenericInterface interfaceValue);

    [Variant]
    public static partial ManagedReferenceGenericUnion<T> Object(
        [UnitySerializeReference] object objectValue);

    [Variant]
    public static partial ManagedReferenceGenericUnion<T> UnityObject(
        [UnitySerializeReference] UnityEngine.Object unityObjectValue);

    [Variant]
    public static partial ManagedReferenceGenericUnion<T> Dictionary(
        [UnitySerializeReference] Dictionary<int, string> dictionaryValue);

    [Variant]
    public static partial ManagedReferenceGenericUnion<T> Multi(
        [UnitySerializeReference] int[,] multiValue);

    [Variant]
    public static partial ManagedReferenceGenericUnion<T> Array(
        [UnitySerializeReference] ManagedReferenceGenericBase[] arrayValues);

    [Variant]
    public static partial ManagedReferenceGenericUnion<T> List(
        [UnitySerializeReference] List<ManagedReferenceGenericBase> listValues);

    [Variant] public static partial ManagedReferenceGenericUnion<T> Empty();
}
