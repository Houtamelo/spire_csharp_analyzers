#nullable enable
using System;
using Houtamelo.Spire;

[Serializable]
[DiscriminatedUnion(Layout.BoxedFields)]
public partial struct UnityBoxedFieldsGeneric<T>
{
    [Variant] public static partial UnityBoxedFieldsGeneric<T> Some(T value);
    [Variant] public static partial UnityBoxedFieldsGeneric<T> None();
}

public static class UnityBoxedFieldsGenericUses
{
    public static UnityBoxedFieldsGeneric<int> IntValue => UnityBoxedFieldsGeneric<int>.Some(42);
    public static UnityBoxedFieldsGeneric<string> StringValue => UnityBoxedFieldsGeneric<string>.Some("text");
    public static UnityBoxedFieldsGeneric<int> IntNone => UnityBoxedFieldsGeneric<int>.None();
    public static UnityBoxedFieldsGeneric<string> StringNone => UnityBoxedFieldsGeneric<string>.None();
}
